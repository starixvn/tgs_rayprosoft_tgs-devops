tar zxvf velero-v1.8.1-linux-amd64.tar.gz
cp velero-v1.8.1-linux-amd64/velero /usr/local/bin


cat <<EOF | sudo tee ~/credentials-velero
[default]
aws_access_key_id=AKIAQ4KFUNHM4PDFPOQH
aws_secret_access_key=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
EOF

BUCKET=tgs-dev-backup
REGION=ap-east-1

velero install \
    --provider aws \
    --plugins velero/velero-plugin-for-aws:v1.4.0 \
    --bucket $BUCKET \
    --backup-location-config region=$REGION \
    --snapshot-location-config region=$REGION \
    --secret-file ./credentials-velero

//Check
velero backup-location get

//建立備份
velero backup create test --exclude-namespaces kube-system,kube-public,kube-node-lease,ingress-nginx,velero

//建立排程
velero schedule create sctest0325 --schedule "*/3 * * * *"
velero schedule create sctest0325 --include-namespace aa --schedule="@daily" --ttl 2160h0m0s
velero schedule create daily-eks --exclude-namespaces kube-system,kube-public,kube-node-lease,ingress-nginx,velero --schedule="@daily" 

//建立還原
velero restore create test1 --from-backup test
velero restore create --from-schedule sc
velero restore create --from-backup daily-eks-20220426000028 
velero restore create --from-backup backup-2 --include-resources persistentvolumeclaims,persistentvolumes
